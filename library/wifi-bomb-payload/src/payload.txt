EXTENSION PASSIVE_RECON

# 1. Variable Definitions
VAR $IFACE = "wlan0"
VAR $MON_IFACE = "wlan0mon"
VAR $PAYLOAD_DIR = "/root/payloads/wifi_bomb"

LOG_INFO "Starting Wi-Fi Bomb Setup..."

# 2. Prepare Required Dependencies and Wireless Interface
FUNCTION PREPARE_ENVIRONMENT()
    # Install Scapy silently if missing and set interface to monitor mode
    STRING python3 -m pip install scapy > /dev/null 2>&1
    STRING airmon-ng start $IFACE > /dev/null 2>&1
    ENTER
END_FUNCTION

# 3. Launch Python Core Script in Background
FUNCTION RUN_CORE()
    # Execute python script in background (nohup / &) with elevated privileges
    STRING nohup python3 $PAYLOAD_DIR/core.py > /dev/null 2>&1 &
    ENTER
END_FUNCTION

# Execution Flow
LED_ATTACK
PREPARE_ENVIRONMENT
DELAY 2000

LOG_INFO "Launching core.py in background..."
RUN_CORE

LED_FINISH
